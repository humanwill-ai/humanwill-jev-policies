cmake_minimum_required(VERSION 3.22)
cmake_policy(SET CMP0091 NEW)
project(humanwill_hook_c VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_NO_SYSTEM_FROM_IMPORTED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(HUMANWILL_FULL_STATIC "Link the Linux executable against musl statically" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_LIBCURL_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_MISC_DOCS OFF CACHE BOOL "" FORCE)
set(CURL_STATIC_CRT ON CACHE BOOL "" FORCE)
set(HTTP_ONLY ON CACHE BOOL "" FORCE)
foreach(feature CURL_ZLIB CURL_BROTLI CURL_ZSTD USE_LIBIDN2 CURL_USE_LIBPSL CURL_USE_LIBSSH2 USE_NGHTTP2 CURL_USE_GSSAPI CURL_USE_LIBUV)
  set(${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(CURL_DISABLE_OPENSSL_AUTO_LOAD_CONFIG ON CACHE BOOL "" FORCE)
if(WIN32)
  set(CURL_USE_SCHANNEL ON CACHE BOOL "" FORCE)
  set(CURL_USE_OPENSSL OFF CACHE BOOL "" FORCE)
else()
  set(CURL_USE_OPENSSL ON CACHE BOOL "" FORCE)
  set(OPENSSL_USE_STATIC_LIBS TRUE)
  if(NOT OPENSSL_ROOT_DIR)
    message(FATAL_ERROR "Provide the pinned static OpenSSL prefix from build_native.py")
  endif()
  if(APPLE)
    set(USE_APPLE_SECTRUST ON CACHE BOOL "" FORCE)
    set(CURL_CA_BUNDLE none CACHE STRING "" FORCE)
    set(CURL_CA_PATH none CACHE STRING "" FORCE)
  else()
    if(NOT HUMANWILL_FULL_STATIC)
      message(FATAL_ERROR "Release Linux build must use musl and HUMANWILL_FULL_STATIC")
    endif()
    set(CURL_CA_BUNDLE /etc/ssl/certs/ca-certificates.crt CACHE STRING "" FORCE)
    set(CURL_CA_PATH /etc/ssl/certs CACHE STRING "" FORCE)
  endif()
endif()
FetchContent_Declare(curl URL https://curl.se/download/curl-8.22.0.tar.xz
  URL_HASH SHA256=f7ef3ae8a22e521f289803fe93543eb64c329b58aa73a9e224dfd915a2a5f4f7
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(curl)
set(SCHEMAS "${CMAKE_CURRENT_BINARY_DIR}/schemas.h")
file(GLOB SCHEMA_INPUTS "${CMAKE_CURRENT_SOURCE_DIR}/../../src/humanwill_policies/schemas/*.json")
add_custom_command(OUTPUT "${SCHEMAS}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_schemas.py" "${SCHEMAS}"
  DEPENDS generate_schemas.py ${SCHEMA_INPUTS})
add_library(hook_json STATIC json.c schema.c platform.c vendor/yyjson/yyjson.c)
target_include_directories(hook_json PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/vendor/yyjson" "${CMAKE_CURRENT_SOURCE_DIR}")
if(WIN32)
  target_link_libraries(hook_json PUBLIC bcrypt)
else()
  if(NOT APPLE)
    find_package(OpenSSL REQUIRED)
    target_link_libraries(hook_json PUBLIC OpenSSL::Crypto m)
  endif()
endif()
add_executable(humanwill-hook-c main.c "${SCHEMAS}")
target_include_directories(humanwill-hook-c PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(humanwill-hook-c PRIVATE hook_json CURL::libcurl)
add_executable(canonical-probe canonical_probe.c)
target_link_libraries(canonical-probe PRIVATE hook_json)
if(HUMANWILL_FULL_STATIC)
  target_link_options(humanwill-hook-c PRIVATE -static)
  target_link_options(canonical-probe PRIVATE -static)
endif()
if(MSVC)
  target_compile_definitions(hook_json PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_definitions(humanwill-hook-c PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(hook_json PRIVATE -Wall -Wextra -Werror -Wno-deprecated-declarations)
  target_compile_options(humanwill-hook-c PRIVATE -Wall -Wextra -Werror -Wno-deprecated-declarations)
endif()
